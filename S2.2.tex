\subsection{Ringed topological spaces}
	
Let $X$ be a topological space.

\begin{exer}
	Let $A$ be a non-trivial abelian group and let $\mathcal A_X$ be the
	constant presheaf of $A$ on $X$. The associated sheaf
	$\mathcal A_X^\dagger$ is the sheaf of locally constant functions valued
	in $A$. In particular, $\mathcal A_X$ is a sheaf if and only if every
	non-empty open subset of $X$ is connected.
\end{exer}
\begin{proof}
	Let $\mathcal F$ be the sheaf of locally constant $A$-valued functions
	on $X$ and let $\varphi: \mathcal A_X \to \mathcal F$ be the evident
	morphism. It is easy to see that the stalks of $\mathcal F$ are all $A$.
	Let $\mathcal G$ be any sheaf and let $\alpha: \mathcal A_X \to
	\mathcal G$ be any morphism. If $f$ is a locally constant function on
	$U$, we may split $U$ into disjoint open sets $U_a = f^{-1}(a)$ for
	$a \in f(U)$. We define $\widetilde{\alpha}(U) : \mathcal F \to
	\mathcal G$ to be the map sending $f$ to the regluing of the sections
	$\alpha(U)(a)|_{U_a}$. It is easy to see that this defines a morphism
	of sheaves and that $\widetilde{\alpha} \circ \varphi = \alpha$. Since
	$\varphi$ is surjective, $\widetilde{\alpha}$ is the unique such
	morphism.
\end{proof}

\begin{exer}
	Let $\mathcal F$ be a sheaf on $X$. Let $s,t \in \mathcal F(X)$. Then
	the set of $x \in X$ such that $s_x = t_x$ is open in $X$.
\end{exer}
\begin{proof}
	If $s_x = t_x$ at some point $x \in X$, there must be a neighborhood
	$U_x$ of $x$ such that $s_y = t_y$ for all $y \in U_x$. Taking the union
	of all such $U_x$ gives the result.
\end{proof}

\begin{exer}
	Let $X$ be a fixed topological space. Let $\mathcal F$ be a presheaf on
	$X$. We keep the notation introduced just before Lemma 2.7.
	\begin{enumerate}[label=(\alph*)]
		\item Let $\mathcal U = \{ U_i \}$ be an open covering of $X$.
			Let ${\cal F_U}(X) = \Ker d_1$. For any open subset $W$
			of $X$, we define in the same way a group
			${\cal F_U}(W)$ relative to the covering
			$\{ W \cap U_i \}$ of $W$. Then ${\cal F_U}$ is a
			presheaf on $X$ and we have a morphism of presheaves
			$\mathcal F \to {\cal F_U}$.
		\item Let $\mathcal V = \{ V_j \}$ be another open covering of
			$X$. We define a partial ordering: $\mathcal V \preceq
			\mathcal U$ if $\mathcal V$ is finer than $\mathcal U$.
			Then there is a canonical homomorphism ${\cal F_U}(W)
			\to {\cal F_V(W)}$ for every open subset $W$ if
			$\mathcal V \preceq \mathcal U$.
		\item The ordering $\preceq$ makes the set of open coverings of
			$X$ into a direct system. Let
			\[
				\mathcal F'(W) =
					\varinjlim_{\mathcal V} {\cal F_V}(W),
			\]
			the direct limit being taken over the open coverings of
			$W$. Then $(\mathcal F')'$ is the sheaf associated to
			$\mathcal F$.
	\end{enumerate}
\end{exer}
\begin{proof}
	Part (a) is clear. The homomorphisms in part (b) are given by the
	resrictions in the natural way.\\

	Part (c) is quite nasty to write out so I won't do it.
\end{proof}

\begin{exer}
	Let $0 \to \mathcal F' \to \mathcal F \to \mathcal F''$ be an exact
	sequence of sheaves on $X$. Then the sequence of abelian groups
	$0 \to \mathcal F'(X) \to \mathcal F(X) \to \mathcal F''(X)$ is exact.
\end{exer}
\begin{proof}
	This follows from the exact sequence induced on stalks.
\end{proof}
