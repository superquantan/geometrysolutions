\subsection{Ringed topological spaces}
	
Let $X$ be a topological space.

\begin{exer}
	Let $A$ be a non-trivial abelian group and let $\mathcal A_X$ be the
	constant presheaf of $A$ on $X$. The associated sheaf
	$\mathcal A_X^\dagger$ is the sheaf of locally constant functions valued
	in $A$. In particular, $\mathcal A_X$ is a sheaf if and only if every
	non-empty open subset of $X$ is connected.
\end{exer}
\begin{proof}
	Let $\mathcal F$ be the sheaf of locally constant $A$-valued functions
	on $X$ and let $\varphi: \mathcal A_X \to \mathcal F$ be the evident
	morphism. It is easy to see that the stalks of $\mathcal F$ are all $A$.
	Let $\mathcal G$ be any sheaf and let $\alpha: \mathcal A_X \to
	\mathcal G$ be any morphism. If $f$ is a locally constant function on
	$U$, we may split $U$ into disjoint open sets $U_a = f^{-1}(a)$ for
	$a \in f(U)$. We define $\widetilde{\alpha}(U) : \mathcal F \to
	\mathcal G$ to be the map sending $f$ to the regluing of the sections
	$\alpha(U)(a)|_{U_a}$. It is easy to see that this defines a morphism
	of sheaves and that $\widetilde{\alpha} \circ \varphi = \alpha$. Since
	$\varphi$ is surjective, $\widetilde{\alpha}$ is the unique such
	morphism.
\end{proof}

\begin{exer}
	Let $\mathcal F$ be a sheaf on $X$. Let $s,t \in \mathcal F(X)$. Then
	the set of $x \in X$ such that $s_x = t_x$ is open in $X$.
\end{exer}
\begin{proof}
	If $s_x = t_x$ at some point $x \in X$, there must be a neighborhood
	$U_x$ of $x$ such that $s_y = t_y$ for all $y \in U_x$. Taking the union
	of all such $U_x$ gives the result.
\end{proof}

\begin{exer}
	Let $X$ be a fixed topological space. Let $\mathcal F$ be a presheaf on
	$X$. We keep the notation introduced just before Lemma 2.7.
	\begin{enumerate}[label=(\alph*)]
		\item Let $\mathcal U = \{ U_i \}$ be an open covering of $X$.
			Let ${\cal F_U}(X) = \Ker d_1$. For any open subset $W$
			of $X$, we define in the same way a group
			${\cal F_U}(W)$ relative to the covering
			$\{ W \cap U_i \}$ of $W$. Then ${\cal F_U}$ is a
			presheaf on $X$ and we have a morphism of presheaves
			$\mathcal F \to {\cal F_U}$.
		\item Let $\mathcal V = \{ V_j \}$ be another open covering of
			$X$. We define a partial ordering: $\mathcal V \preceq
			\mathcal U$ if $\mathcal V$ is finer than $\mathcal U$.
			Then there is a canonical homomorphism ${\cal F_U}(W)
			\to {\cal F_V(W)}$ for every open subset $W$ if
			$\mathcal V \preceq \mathcal U$.
		\item The ordering $\preceq$ makes the set of open coverings of
			$X$ into a direct system. Let
			\[
				\mathcal F'(W) =
					\varinjlim_{\mathcal V} {\cal F_V}(W),
			\]
			the direct limit being taken over the open coverings of
			$W$. Then $(\mathcal F')'$ is the sheaf associated to
			$\mathcal F$.
	\end{enumerate}
\end{exer}
\begin{proof}
	Part (a) is clear. The homomorphisms in part (b) are given by the
	resrictions in the natural way.\\

	Part (c) is quite nasty to write out so I won't do it.
\end{proof}

\begin{exer}
	Let $0 \to \mathcal F' \to \mathcal F \to \mathcal F''$ be an exact
	sequence of sheaves on $X$. Then the sequence of abelian groups
	$0 \to \mathcal F'(X) \to \mathcal F(X) \to \mathcal F''(X)$ is exact.
\end{exer}
\begin{proof}
	This follows from the exact sequence induced on stalks.
\end{proof}

\begin{exer}
	Let $f: X \to Y, g: Y \to Z$ be continuous maps, $\mathcal F$ a sheaf on
	$X$ and $\mathcal H$ a sheaf on $Z$. Then
	\[
		f^{-1}(g^{-1}\mathcal H) = (g \circ f)^{-1}\mathcal H,\,
		g_*(f_*\mathcal F) = (g \circ f)_*\mathcal F.
	\]
\end{exer}
\begin{proof}
	The second part is obvious. For the first one, notice that the
	assignment
	\[
		\mathcal F \mapsto \left(U \mapsto
			\varprojlim_{V \supseteq f(U)}\mathcal F(U)\right)
	\]
	is functorial, so we have a morphism $f^{-1}(g^{-1}\mathcal H) \to
	(g \circ f)^{-1}\mathcal H$. It is easy to check that this restricts to
	isomorphisms on the stalks, so it is an isomorphism. We can say this is
	equality because the isomorphism is sufficiently special.
\end{proof}

\begin{exer}
	Let $\mathcal B$ be a base of open subsets for a topological space $X$.
	Let $\mathcal F, \mathcal G$ be two sheaves on $X$. Suppose that for
	every $U \in \mathcal B$, there is a homomorphism
	$\alpha(U): \mathcal F(U) \to \mathcal G(U)$ which is compatible with
	restrictions. Then there is a unique extension $\alpha: \mathcal F \to
	\mathcal G$ to a homomorphism of sheaves. Moreover, if $\alpha(U)$ is
	surjective (resp. injective) for every $U \in \mathcal B$, then $\alpha$
	is surjective (resp. injective).
\end{exer}
\begin{proof}
	Let $U$ be any open set and $s$ a section of $\mathcal F$ over $U$. If
	$\{U_i\}$ is a cover of $U$ by elements of $\mathcal B$, we have a
	corresponding family $\{s_i\}$ of sections of $\mathcal F$ given by
	restrictions of $s$. Since $\alpha$ is compatible with restrictions, we
	obtain a family $\{\alpha(U_i)(s_i)\}$ that agrees on intersections,
	which reglues into a section $\alpha(s)$ of $\mathcal G$. It is easy to
	see that this is a morphism of sheaves, and the uniqueness of this
	extension follows from the uniqueness of gluing. The statement about
	surjectivity and injectivity is obvious.
\end{proof}

\begin{exer}
	Let $\{U_i\}$ be an open covering of $X$. Suppose we are given a family
	of sheaves $\mathcal F_i$ on $U_i$ and isomorphisms $\phi_{ij}:
	{\mathcal F_i|}_{U_i \cap U_j} \to {\mathcal F_j|}_{U_i \cap U_j}$ such
	that $\phi_{ik} = \phi_{jk} \circ \phi_{ij}$ on $U_i \cap U_j \cap U_k$ 
	and $\phi{ii} = 1_{\mathcal F_i}$. Then there is a unique sheaf
	$\mathcal F$ on $X$ with isomorphisms $\psi_i: {\mathcal F|}_{U_i} \to
	\mathcal F_i$ such that $\psi_j = \phi_{ij} \circ \psi_i$ on $U_i \cap
	U_j$.
\end{exer}
\begin{proof}
	The sheaf $\mathcal F$ is quite simple in its definition. For any open
	subset $V$ of $X$, we define a section of $\mathcal F(V)$ to be a family
	$\{s_i\}$ such that $s_i \in \mathcal F_i(V \cap U_i)$ and ${s_i|}_{V
	\cap U_i \cap U_j} = \phi_{ji}({s_j|}_{V \cap U_i \cap U_j})$ for all
	$i,j$. It is easy to see that this defines a sheaf, and the existence of
	the desired isomorphisms is clear. To see that it is unique, let
	$\mathcal G$ be another such sheaf, and denote the isomorphisms
	${\mathcal G|}_{U_i} \to \mathcal F_i$ by $\psi'$. Then we have a family
	of isomorphisms ${\mathcal G|}_{U_i} \to {\mathcal F|}_{U_i}$, which
	we may glue to an isomorphism $\mathcal G \to \mathcal F$.
\end{proof}

\begin{exer}
	Let $\mathcal F$ be a sheaf on $X$ whose support $Y$ is a finite set of
	closed points $y$. Then for any open set $U$ of $X$, $\mathcal F =
	\oplus_{x \in U \cap Y} \mathcal F_x$.
\end{exer}
\begin{proof}
	Let $U$ be an open subset of $X$ meeting $Y$ at only a single point $y$.
	Then for any open neighborhood $V$ of $y$ contained in $U$, we have $U
	= (U \setminus \{y\}) \cup V$. Since $\mathcal F(U \setminus \{y\}) =
	0$, the value of any section $s \in \mathcal F(U)$ is determined by its
	restriction to $V$, hence by its stalk at $y$. The statement now follows
	from an easy gluing argument.
\end{proof}
