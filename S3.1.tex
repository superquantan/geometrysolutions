\subsection{The technique of base change}

\begin{exer}
	Let $S$, $X$, and $Y$ be sets, and equip $X$ (resp. $Y$) with a map
	$\varphi: X \to S$ (resp. $\psi\colon Y \to S$). Let
	$p: X \times_S Y \to X$ and $q: X \times_S Y \to Y$ be the
	restrictions of the projections from $X \times Y$. Then for any set $Z$,
	equipped with a pair of maps $f: Z \to X$ and $g: Z \to Y$
	such that $\phi \circ f = \psi \circ g$, there is a unique map
	$(f,g)_S: Z \to X \times_S Y$ such that $p \circ (f,g)_S = f$ and
	$q \circ (f,g)_S = g$.
\end{exer}
\begin{proof}
	The existence (but perhaps not the uniqueness) of $(f,g)_S$ is clear, so
	suppose that $h: Z \to X \times_S Y$ is such that $p \circ h = f$
	and $q \circ h = g$. Then for all $z \in Z$, $h(x) = (p(h(x)),q(h(x)))
	= (f(x),g(x)) = (f,g)_S(x)$ and so $h = (f,g)_S$.
\end{proof}

\begin{exer}
	Let $f: X \to Y$ be a morphism of schemes. For any scheme $T$, let
	$f(T): X(T) \to Y(T)$ denote the map defined by $f(T)(g) =
	f \circ g$. Then $f(T)$ is bijective for every $T$ if and only if
	$f$ is an isomorphism.
\end{exer}
\begin{proof}
	Yoneda
\end{proof}

\begin{exer}
	Let $f: X \to Y$ be a dominant morphism of irreducible schemes. Then
	the generic fiber of $f$ is dense in $X$.
\end{exer}
\begin{proof}
	Let $\zeta_X$ and $\zeta_Y$ be the generic points of $X$ and $Y$,
	respectively. Then Proposition 1.16 implies that $\zeta_X$ is in the
	generic fiber, so it is dense.
\end{proof}

\begin{exer}
	Let $f: X \to Y$ be a morphism of affine schemes. Then for any affine
	open subscheme $V$ of $Y$, $f^{-1}(V)$ is affine.
\end{exer}
\begin{proof}
	Just recall that $f^{-1}(V) = X \times_Y V$, so is affine.
\end{proof}

\begin{exer}
	Let $X$ and $Y$ be algebraic varieties over a field $k$. Then
	\[
		\dim (X \times_k Y) = \dim X + \dim Y.
	\]
\end{exer}
\begin{proof}
	It suffices to prove this for affine varieties. Suppose that $X = \Spec
	A$ and $Y = \Spec B$. By Corollary 2.5.10, $\dim X = \trdeg_k A$ and
	$\dim Y = \trdeg_k B$. The usual sorites of tensor products give the
	result.
\end{proof}

\begin{exer}
	Let $\pi: T \to S$ be a morphism of schemes. Assume that $\pi$ is either
	an open immersion, closed immersion, or that $S = \Spec A$ and $\pi$ is
	induced by a localization $A \to F^{-1}A$ for a multiplicative set $F
	\subset A$.
	\begin{enumerate}[label=(\alph*)]
		\item Let $f$,$g: Z \to T$ be two morphisms of schemes such that
			$\pi \circ f = \pi \circ g$. Then $f = g$.
		\item Let $X$, $Y$ be $T$-schemes. Then the canonical morphism
			$\varphi: X \times_T Y \to X \times_S Y$ is an
			isomorphism.
		\item Part (b) is not always true when $\pi$ is not a
			monomorphism.
	\end{enumerate}
\end{exer}
\begin{proof}
	It is clear that $\pi$ is a monomorphism, so (a) is immediate. For
	(b), recall that for any scheme $Z$, $(X \times_T Y)(Z) =
	X(Z) \times_{T(Z)} Y(Z)$ and similarly for $(X \times_S Y)(Z)$. It then
	suffices to prove the statement for sets, in which case the statement
	is clear. Finally, let $S = \Spec \FF_2$, $T = \Spec \ZZ$, and $X = Y =
	\Spec \ZZ[T]$. Then $X \times_T Y = \Spec \ZZ[T_1,T_2]$ and $X \times_S
	Y = \Spec \FF_2[T_1,T_2]$, which are clearly not isomorphic.
\end{proof}

\begin{exer}
	Let $X$, $Y$ be $S$-schemes, $p$ and $q$ the projection morphisms from
	$X \times_S Y$ to $X$ and $Y$. Fix $s \in S$. Then for any $x \in X_s$
	and $y \in Y_s$, there is a natural homeomorphism
	\[
		\Spec(\kappa(x) \otimes_{\kappa(s)} \kappa(y))
		\to Z = \{z \in X \times_S Y | p(z) = x,\, q(z) = y\}.
	\]
\end{exer}
\begin{proof}
	Consider the fibered product
	$X \times_S Y \times_S \Spec \kappa(s) \times_X \Spec \kappa(x)
	\times_Y \kappa(y)$. It is easy to see that this is isomorphic to
	$(X_s \times_S Y_s) \times_X \Spec \kappa(x) \times_Y \Spec \kappa(y)$.
	and homeomorphic to $Z$. Since the morphism out of $\Spec \kappa(x)$
	factors through $X_s$, we have that this product is isomorphic to $Y_s
	\times_S \Spec \kappa(x) \times_Y \Spec \kappa(y).$ Similar reasoning
	means that this product is then isomorphic to $\Spec \kappa(x) \times_S
	\Spec \kappa(y)$. Finally, these both factor through $\Spec \kappa(s)$,
	so the product is just $\Spec(\kappa(x) \otimes_{\kappa(s)} \kappa(y))$.
\end{proof}
